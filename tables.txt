create table if not exists FoodItems (
  id bigint generated by default as identity primary key,
  food_name varchar not null,
  category varchar not null check (
    category in ('Appetizer', 'Main Course', 'Salad', 'Desert', 'Beverage')
  ),
  description varchar,
  cooking_time varchar,
  img_url text,
  old_price float default 0,
  new_price float not null,
  vegeterian boolean default false,
  gluten_free boolean default false,
  protin_rich boolean default false,
  dairy boolean default false,
  eggs boolean default false,
  nuts boolean default false,
  created_at timestamp with time zone default now()
);
